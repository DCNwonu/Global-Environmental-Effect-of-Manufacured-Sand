# -----------------------
# Helpers
# -----------------------
def interp_zero(x1, y1, x2, y2):
    """
    Linear interpolation to find x where y=0 between (x1,y1) and (x2,y2).
    Returns None if no sign change or invalid values.
    """
    if np.isnan(y1) or np.isnan(y2):
        return None
    if y1 == 0:
        return x1
    if y2 == 0:
        return x2
    if y1 * y2 > 0:   # same sign -> no crossing
        return None
    if y1 == y2:
        return None
    return x1 + (0 - y1) * (x2 - x1) / (y2 - y1)

def format_km(km):
    if km is None or (isinstance(km, float) and np.isnan(km)):
        return "no flip"
    if km < 1000:
        return f"{km:.0f} km"
    if km < 10000:
        return f"{km/1000:.2f}k km"
    return f"{km/1000:.1f}k km"

def annotate_threshold(ax, d0, text, y_frac):
    """
    Place a readable annotation in axes coordinates (not on the x-axis),
    with arrow to the (d0,0) crossing.
    """
    ax.scatter([d0], [0], marker="x", s=55, color="black", zorder=5)
    ax.annotate(
        text,
        xy=(d0, 0), xycoords="data",
        xytext=(0.02, y_frac), textcoords="axes fraction",
        ha="left", va="center",
        bbox=dict(boxstyle="round,pad=0.25", fc="white", ec="0.6", alpha=0.92),
        arrowprops=dict(arrowstyle="->", lw=0.8, color="0.3"),
        fontsize=9
    )

# -----------------------
# Load workbook as raw grid
# -----------------------
raw = pd.read_excel(XLSX_PATH, header=None)

# distance extremes are in rows 4 and 5, col 0 (0-indexed)
dist1 = float(raw.iloc[4, 0])
dist2 = float(raw.iloc[5, 0])
x = np.array([dist1, dist2], dtype=float)

# -----------------------
# PB for climate change block (cols 1–3)
# -----------------------
pb_cols = [1, 2, 3]
pb_labels = [str(raw.iloc[3, c]).strip() for c in pb_cols]
pb_vals = np.array([
    [pd.to_numeric(raw.iloc[4, c], errors="coerce"),
     pd.to_numeric(raw.iloc[5, c], errors="coerce")]
    for c in pb_cols
], dtype=float)

pb_cross = []
for lab, (y1, y2) in zip(pb_labels, pb_vals):
    pb_cross.append({
        "subscenario": lab,
        "y_at_dist1": float(y1),
        "y_at_dist2": float(y2),
        "d0_km": interp_zero(dist1, y1, dist2, y2)
    })
pb_cross = pd.DataFrame(pb_cross)
pb_cross.to_csv(os.path.join(OUT_DIR, "PB_climate_zero_crossings.csv"), index=False)

# -----------------------
# RCB block parser
# Your layout:
#   row 1: summary (Min/Median/Max) in first col of each group
#   row 2: probability (p=17, p=83)
#   row 3: bound label (lower bound (%), upper bound (%))
#   rows 4-5: values at the two distance extremes
# -----------------------
def parse_rcb_block(start_col, end_col):
    records = []
    last_summary = None

    for c in range(start_col, end_col + 1):
        summary = raw.iloc[1, c]   # may be blank except first col of group
        prob = raw.iloc[2, c]
        bound = raw.iloc[3, c]

        # forward fill summary
        if pd.notna(summary) and str(summary).strip().lower() in ["min", "median", "max"]:
            last_summary = str(summary).strip()
        if last_summary is None:
            continue

        if pd.isna(prob) or pd.isna(bound):
            continue

        y1 = pd.to_numeric(raw.iloc[4, c], errors="coerce")
        y2 = pd.to_numeric(raw.iloc[5, c], errors="coerce")
        if pd.isna(y1) or pd.isna(y2):
            continue

        records.append({
            "summary": last_summary,
            "prob": str(prob).strip(),
            "bound": str(bound).strip(),
            "y_at_dist1": float(y1),
            "y_at_dist2": float(y2),
            "d0_km": interp_zero(dist1, float(y1), dist2, float(y2))
        })

    return pd.DataFrame(records)

# Column blocks from your file
rcb15 = parse_rcb_block(5, 16)   # 1.5 °C block
rcb20 = parse_rcb_block(18, 29)  # 2.0 °C block

rcb15.to_csv(os.path.join(OUT_DIR, "RCB15_zero_crossings.csv"), index=False)
rcb20.to_csv(os.path.join(OUT_DIR, "RCB20_zero_crossings.csv"), index=False)

# -----------------------
# Plot 2 & 3: RCB blocks (1.5 °C and 2.0 °C)
# -----------------------
def plot_rcb_panels(df, title, outname):
    summaries = ["Min", "Median", "Max"]
    fig, axes = plt.subplots(1, 3, figsize=(16.8, 5.2), sharey=True)

    for ax, summ in zip(axes, summaries):
        sub = df[df["summary"].str.lower() == summ.lower()].copy()
        ax.axhline(0, color="0.5", lw=1.0)

        # Plot each line and store annotations
        ann_items = []
        for _, r in sub.iterrows():
            y = np.array([r["y_at_dist1"], r["y_at_dist2"]], float)
            label = f"{r['prob']} {r['bound']}"
            ax.plot(x, y, marker="o", lw=1.5, label=label)
            if r["d0_km"] is not None:
                ann_items.append((r["d0_km"], f"{label}: 0% @ {format_km(r['d0_km'])}"))

        # Stack annotation boxes neatly
        for j, (d0, txt) in enumerate(ann_items):
            y_frac = 0.90 - 0.12 * j
            if y_frac < 0.18:
                break
            annotate_threshold(ax, d0, txt, y_frac)

        ax.set_xscale("log")
        ax.set_title(summ)
        ax.set_xlabel("Import distance (km) [log scale]")
        ax.grid(True, which="both", axis="x", alpha=0.2)
        ax.grid(True, axis="y", alpha=0.2)
        ax.legend(frameon=False, fontsize=10, loc="lower right")

        if ax is axes[0]:
            ax.set_ylabel("Relative perturbation effect (Δ%)")

    fig.suptitle(title, y=1.03, fontsize=22)
    fig.tight_layout()
    fig.savefig(os.path.join(OUT_DIR, outname), dpi=DPI, bbox_inches="tight")
    plt.show()

plot_rcb_panels(
    rcb15,
    "B) 1.5 °C RCB: transport distance threshold",
    "Fig_RCB15_thresholds_v2.png"
)

plot_rcb_panels(
    rcb20,
    "C) 2.0 °C RCB: transport distance threshold",
    "Fig_RCB20_thresholds_v2.png"
)
