def compute_delta_percent(df: pd.DataFrame) -> pd.DataFrame:
    """
    Compute Δ% (manufactured vs imported).
    Imported has no uncertainty (fixed); manufactured has min/max.
    """
    df = df.copy()
    I = pd.to_numeric(df["imported_median"], errors="coerce")

    if I.isna().any():
        raise ValueError("imported_median contains missing or non-numeric values.")
    if (I == 0).any():
        bad = df.loc[I == 0, ["family", "indicator", "region"]]
        raise ValueError(f"Imported median is zero for some rows; %Δ undefined:\n{bad}")

    for col in ["manufactured_median", "manufactured_min", "manufactured_max"]:
        df[col] = pd.to_numeric(df[col], errors="coerce")
        if df[col].isna().any():
            raise ValueError(f"{col} contains missing or non-numeric values.")

    df["delta_pct_med"] = (df["manufactured_median"] - I) / I * 100.0
    df["delta_pct_min"] = (df["manufactured_min"] - I) / I * 100.0
    df["delta_pct_max"] = (df["manufactured_max"] - I) / I * 100.0

    return df


def nature_forest_delta_percent(
    df_in: pd.DataFrame,
    family_order,
    indicator_order_by_family,
    region_order=("Canada", "China", "RoW"),
    panel_letters=("A", "B", "C"),
):
    """
    Nature-style 3-panel Δ% forest plot.
    - One subplot per family (A–C)
    - Points = median Δ% ; whiskers = min–max Δ%
      (uncertainty from manufactured only; imported fixed)
    - Consistent marker+color per region across all panels
    - Single legend (figure-level)
    """
    df = compute_delta_percent(df_in)
    
    region_style = {
        "Canada": {"marker": "o", "color": "C0"},
        "China":  {"marker": "^", "color": "C1"},
        "RoW":    {"marker": "s", "color": "C2"},
    }

    offsets = {"Canada": -0.18, "China": 0.00, "RoW": 0.18}

    plt.rcParams.update({
        "font.size": 15,
        "axes.titlesize": 15,
        "axes.labelsize": 15,
        "legend.fontsize": 15,
        "legend.title_fontsize": 15,
    })

    fig, axes = plt.subplots(
        nrows=1, ncols=len(family_order),
        figsize=(12, 4.8),
        sharex=True,
        gridspec_kw={"wspace": 0.25}
    )

    xlim = (-100, 800)

    for ax_i, fam in enumerate(family_order):
        ax = axes[ax_i] if len(family_order) > 1 else axes
        fam_df = df[df["family"] == fam].copy()

        inds = indicator_order_by_family[fam]
        y_positions = {ind: (len(inds) - 1 - j) for j, ind in enumerate(inds)}

        for reg in region_order:
            style = region_style[reg]
            sub = fam_df[fam_df["region"] == reg]

            for _, r in sub.iterrows():
                ind = r["indicator"]
                if ind not in y_positions:
                    continue

                y0 = y_positions[ind] + offsets[reg]
                x = float(r["delta_pct_med"])
                lo = float(r["delta_pct_min"])
                hi = float(r["delta_pct_max"])
                xerr = np.array([[x - lo], [hi - x]]) 

                ax.errorbar(
                    x, y0, xerr=xerr,
                    fmt=style["marker"],
                    color=style["color"],
                    markersize=4.8,
                    capsize=2.5,
                    linewidth=1.0,
                )

        ax.axvline(0, linewidth=1.0, color="0.2")

        ax.set_yticks([y_positions[ind] for ind in inds])
        ax.set_yticklabels(inds)

        ax.set_xlim(*xlim)
        ax.grid(True, axis="x", alpha=0.25)

        ax.set_title(f"{panel_letters[ax_i]}  {fam}", pad=20)

        ax.spines["top"].set_visible(False)
        ax.spines["right"].set_visible(False)

        if ax_i == 0:
            ax.set_ylabel("Indicator")
        else:
            ax.set_ylabel("")

    fig.supxlabel("Δ% impact (manufactured vs imported)")

    handles = []
    labels = []
    for reg in region_order:
        style = region_style[reg]
        h = plt.Line2D(
            [], [], linestyle="None",
            marker=style["marker"],
            color=style["color"],
            markersize=6,
            label=reg
        )
        handles.append(h)
        labels.append(reg)

    fig.legend(
        handles, labels, title="Region",
        loc="upper center", ncol=3,
        frameon=False, bbox_to_anchor=(0.5, 0.98)
    )

    fig.subplots_adjust(bottom=0.19, left=0.08, right=0.98, top=0.70, wspace=0.25)

nature_forest_delta_percent(
