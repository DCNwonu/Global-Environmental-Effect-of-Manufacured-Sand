records = []
for region, scenarios in metrics.items():
    for scenario, years_data in scenarios.items():
        for year, countries in years_data.items():
            for country, metric_vals in countries.items():
                for metric, value in metric_vals.items():
                    records.append({
                        "Region": region,
                        "Scenario": scenario,
                        "Year": year,
                        "Country": country,
                        "Metric": metric,
                        "Metric_Value": value
                    })

df_long = pd.DataFrame(records)

df_filtered = df_long[df_long['Metric'].isin(['Imported', 'Manufactured'])]
regions = df_filtered['Region'].unique()
years = sorted(df_filtered['Year'].unique())

metric_colors = {
    'Imported': 'royalblue',
    'Manufactured': 'orange'
}
fig = make_subplots(
    rows=1, cols=3,
    shared_yaxes=True,
    subplot_titles=regions
)
for col_idx, region in enumerate(regions, start=1):
    df_region = df_filtered[df_filtered['Region'] == region]

    for metric in ['Imported', 'Manufactured']:
        df_best = df_region[df_region['Scenario'] == 'Best Case']
        df_prob = df_region[df_region['Scenario'] == 'Probable Case']
        df_worst = df_region[df_region['Scenario'] == 'Worst Case']

        df_merge = df_prob.merge(
            df_best[['Year', 'Country', 'Metric', 'Metric_Value']],
            on=['Year', 'Country', 'Metric'],
            suffixes=('_prob', '_best')
        ).merge(
            df_worst[['Year', 'Country', 'Metric', 'Metric_Value']],
            on=['Year', 'Country', 'Metric']
        )

        df_merge = df_merge[df_merge['Metric'] == metric]

        if region == "RoW":
            df_year = df_merge.copy()
        else:
            df_year = df_merge.groupby('Year').sum(numeric_only=True).reset_index()

        probable_vals = df_year['Metric_Value_prob']
        best_vals = df_year['Metric_Value_best']
        worst_vals = df_year['Metric_Value']

        upper_error = worst_vals - probable_vals
        lower_error = probable_vals - best_vals

        if region == "RoW":
            x_vals = df_year['Year'].astype(str) + " - " + df_year['Country']
        else:
            x_vals = [str(y) for y in years]

        fig.add_trace(
    go.Bar(
                x=x_vals,
                y=probable_vals,
                name=metric,
                marker_color=metric_colors[metric],
                error_y=dict(
                    type='data',
                    symmetric=False,
                    array=upper_error,
                    arrayminus=lower_error
                ),
                showlegend=(col_idx == 1)
            ),
            row=1, col=col_idx
        )
fig.update_layout(
    title="kgN",
    barmode='group',
    height=500,
    width=1200,
    margin=dict(t=80, l=50, r=50, b=50)
)

fig.update_yaxes(tickformat=".2e")

fig.update_xaxes(title_text="Year", row=1, col=1)
fig.update_xaxes(title_text="Year", row=1, col=2)
fig.update_xaxes(title_text="Year", row=1, col=3)

fig.update_yaxes(title_text="Tg N/yr", row=1, col=1)
fig.show()
