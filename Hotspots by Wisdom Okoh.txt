countries = data["Country"].unique().tolist()
hotspots = data["Hotspot"].unique().tolist()
impacts = data["Impact"].unique().tolist()

labels = countries + hotspots + impacts
label_dict = {label: i for i, label in enumerate(labels)}

source_1 = data["Country"].map(label_dict)
target_1 = data["Hotspot"].map(label_dict)

source_2 = data["Hotspot"].map(label_dict)
target_2 = data["Impact"].map(label_dict)

node_colors = []
for label in labels:

        node_colors.append("#E0E0E0")
