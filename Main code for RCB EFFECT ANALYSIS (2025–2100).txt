import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.colors as mcolors
from scipy.optimize import root_scalar
from matplotlib.lines import Line2D

# ----------------------------
# USER SETTINGS
# ----------------------------
INPUT_XLSX = "/content/technology_shift_pb_evaluation_outputs_publication.xlsx"
SHEET_ANNUAL_WORLD = "Annual_WORLD"

OUT_DIR = "/content/rcb_outputs"
os.makedirs(OUT_DIR, exist_ok=True)

SAVE_FIGS = True
DPI = 1200

# Carbon budget baseline model (exponential pathway)
t0 = 2022
years_projection = np.arange(2022, 2101, 1)
y0_projection = 41.5  # GtCO2/yr in 2022

RCBs = np.array([2000, 1450, 1150, 950, 800], dtype=float)     # GtCO2
probabilities = np.array([17, 33, 50, 67, 83], dtype=int)       # %

SSP_ORDER = ["SSP1", "SSP2", "SSP3", "SSP4", "SSP5"]
SSP_COLORS = {"SSP1":"C0","SSP2":"C1","SSP3":"C2","SSP4":"C3","SSP5":"C4"}

# ----------------------------
# FIGURE STYLE
# ----------------------------
def set_style():
    plt.rcParams.update({
        "font.family": "sans-serif",
        "font.sans-serif": ["Arial", "DejaVu Sans"],
        "font.size": 11,
        "axes.titlesize": 12,
        "axes.labelsize": 11,
        "xtick.labelsize": 10,
        "ytick.labelsize": 10,
        "legend.fontsize": 10,
        "legend.title_fontsize": 10,
        "axes.spines.top": False,
        "axes.spines.right": False,
        "figure.dpi": DPI,
        "savefig.dpi": 1200,
    })

def savefig(fig, filename):
    if SAVE_FIGS:
        path = os.path.join(OUT_DIR, filename)
        fig.savefig(path, bbox_inches="tight", dpi=DPI)
        print(f"Saved: {path}")
        return path
    return None

# ----------------------------
# 1) Load Annual_WORLD (WORLD, climate), compute cumulative Δ since 2025
# ----------------------------
annual_world = pd.read_excel(INPUT_XLSX, sheet_name=SHEET_ANNUAL_WORLD)

clim = annual_world[
    (annual_world["region"] == "WORLD") &
    (annual_world["pb_indicator"] == "Climate change (CO2 emissions)")
].copy()

clim = clim[(clim["year"] >= 2025) & (clim["year"] <= 2100)].copy()
clim["delta_footprint"] = pd.to_numeric(clim["delta_footprint"], errors="coerce")
clim = clim.dropna(subset=["delta_footprint"]).copy()

clim["ssp"] = clim["ssp"].astype(str).str.strip()
clim["uncertainty_stat"] = clim["uncertainty_stat"].astype(str).str.strip().str.lower()
clim["scenario_variant"] = clim["scenario_variant"].astype(str).str.strip()

# Six scenario variants 
scenario_variants = sorted(clim["scenario_variant"].unique())[:6]

# cumulative Δ emissions since 2025
clim = clim.sort_values(["scenario_variant","ssp","uncertainty_stat","year"]).copy()
clim["cum_delta_gtco2_2025"] = clim.groupby(["scenario_variant","ssp","uncertainty_stat"])["delta_footprint"].cumsum()

# ----------------------------
# 2) Baseline annual pathway E_p(t) per probability and remaining budget
# ----------------------------
def cumulative_emissions(k, years, y0):
    y = y0 * np.exp(-k * (years - years[0]))
    return np.trapz(y, years)

baseline = {}  # prob -> dict(rcb, year, E, cumE, remaining)
for prob, rcb in zip(probabilities, RCBs):
    f = lambda k: cumulative_emissions(k, years_projection, y0_projection) - rcb
    k = root_scalar(f, bracket=[1e-8, 5.0], method="brentq").root

    E = y0_projection * np.exp(-k * (years_projection - years_projection[0]))  # baseline annual emissions
    cumE = np.zeros_like(E, dtype=float)
    for i in range(1, len(years_projection)):
        cumE[i] = np.trapz(E[:i+1], years_projection[:i+1])

    remaining = rcb - cumE

    baseline[prob] = {"rcb": float(rcb), "year": years_projection, "E": E, "cumE": cumE, "remaining": remaining}

def baseline_remaining_series(prob):
    df = pd.DataFrame({"year": baseline[prob]["year"], "remaining": baseline[prob]["remaining"]})
    df = df[(df["year"] >= 2025) & (df["year"] <= 2100)].copy()
    return df.set_index("year")["remaining"]

def baseline_emissions_series(prob):
    df = pd.DataFrame({"year": baseline[prob]["year"], "E": baseline[prob]["E"]})
    df = df[(df["year"] >= 2025) & (df["year"] <= 2100)].copy()
    return df.set_index("year")["E"]

# ----------------------------
# 3) Adjusted annual emissions and adjusted remaining budget
#   E_adj(t) = E_base(t) + ΔE(t)
#   RCB_adj(t) = R_p - ∫ E_adj = RCB_base(t) - ∫ ΔE
# ----------------------------
rows = []
for prob in probabilities:
    rcb_val = baseline[prob]["rcb"]
    base_rem = baseline_remaining_series(prob)
    base_E = baseline_emissions_series(prob)

    for (scen, ssp, ustat), sub in clim.groupby(["scenario_variant","ssp","uncertainty_stat"]):
        if scen not in scenario_variants:
            continue
        sub = sub.sort_values("year")
        yrs = sub["year"].to_numpy(int)

        dE = sub["delta_footprint"].to_numpy(float)            # ΔE(t)
        cum_dE = sub["cum_delta_gtco2_2025"].to_numpy(float)   # ∫_{2025..t} ΔE

        E_base = base_E.loc[yrs].to_numpy(float)
        E_adj = E_base + dE

        R_base = base_rem.loc[yrs].to_numpy(float)
        R_adj = R_base - cum_dE  # remaining budget after adding ΔE

        rows.append(pd.DataFrame({
            "probability_pct": prob,
            "RCB_gtco2": rcb_val,
            "scenario_variant": scen,
            "ssp": ssp,
            "uncertainty_stat": ustat,
            "year": yrs,
            "deltaE_gtco2yr": dE,
            "E_base_gtco2yr": E_base,
            "E_adj_gtco2yr": E_adj,
            "cum_deltaE_gtco2_2025": cum_dE,
            "RCB_base_remaining_gtco2": R_base,
            "RCB_adj_remaining_gtco2": R_adj,
        }))

effect_df = pd.concat(rows, ignore_index=True)

# ============================================================
# FIGURE A: cumulative Δ emissions panels
# ============================================================
set_style()
fig, axes = plt.subplots(2, 3, figsize=(14.6, 7.6), sharex=True)
axes = axes.ravel()

for ax_i, scen in enumerate(scenario_variants):
    ax = axes[ax_i]
    sub = clim[clim["scenario_variant"] == scen].copy()

    for ssp in SSP_ORDER:
        ssp_sub = sub[sub["ssp"] == ssp].copy()
        if ssp_sub.empty:
            continue

        piv = ssp_sub.pivot(index="year", columns="uncertainty_stat", values="cum_delta_gtco2_2025")
        if {"min","median","max"}.issubset(piv.columns):
            ax.fill_between(piv.index, piv["min"], piv["max"], color=SSP_COLORS[ssp], alpha=0.10, linewidth=0)
            ax.plot(piv.index, piv["median"], color=SSP_COLORS[ssp], lw=1.3)
        else:
            ax.plot(ssp_sub["year"], ssp_sub["cum_delta_gtco2_2025"], color=SSP_COLORS[ssp], lw=1.0)

    ax.axhline(0, lw=0.8, color="0.6")
    ax.set_title(f"{chr(65+ax_i)}  {scen}")
    ax.grid(True, axis="y", alpha=0.2)

    if ax_i in [0, 3]:
        ax.set_ylabel("Cumulative Δ emissions since 2025 (GtCO₂)")
    if ax_i in [3, 4, 5]:
        ax.set_xlabel("Year")


fig.subplots_adjust(top=0.88, right=0.82, wspace=0.28, hspace=0.35)
savefig(fig, "FigA_cumDelta_SSP1to5.png")
plt.show()

# ============================================================
# FIGURE B: heatmap panels (min/median/max across scenarios)
# ============================================================
cum2100 = clim[clim["year"] == 2100].copy()
cum2100["effect_on_RCB_gtco2"] = -cum2100["cum_delta_gtco2_2025"]

agg = cum2100.groupby("scenario_variant")["effect_on_RCB_gtco2"].agg(["min", "median", "max"]).reset_index()

# ---- LABEL REPLACEMENT (display only) ----
label_map = {
    "Best case": "Lower bound",
    "Worst case": "Upper bound",
}

def rename_case_labels(label: str) -> str:
    label = str(label)
    for src, dst in label_map.items():
        if label.startswith(src):
            return label.replace(src, dst, 1)
    return label

scenario_order = scenario_variants  # keep original ordering
display_columns = [rename_case_labels(s) for s in scenario_order]

heatmaps = {}
for stat in ["min", "median", "max"]:
    mat = []
    for prob, rcb in zip(probabilities, RCBs):
        row = []
        for scen in scenario_order:
            val_gt = float(agg.loc[agg["scenario_variant"] == scen, stat].iloc[0])
            row.append(val_gt / rcb * 100.0)
        mat.append(row)

    heatmaps[stat] = pd.DataFrame(
        mat,
        index=[f"p={int(p)}%" for p in probabilities],
        columns=display_columns
    )

set_style()
fig, axes = plt.subplots(1, 3, figsize=(16, 4.8), sharey=True)

all_vals = np.concatenate([heatmaps[s].to_numpy().ravel() for s in ["min", "median", "max"]])
vmax = np.nanpercentile(np.abs(all_vals), 99)
vmax = max(vmax, 0.01)
norm = mcolors.TwoSlopeNorm(vmin=-vmax, vcenter=0, vmax=vmax)

titles = ["Min across scenarios", "Median across scenarios", "Max across scenarios"]
stats = ["min", "median", "max"]

im = None
for ax, stat, ttl in zip(axes, stats, titles):
    mat = heatmaps[stat]
    im = ax.imshow(mat.values, cmap="coolwarm", norm=norm, aspect="auto")
    ax.set_title(ttl)
    ax.set_xticks(np.arange(mat.shape[1]))
    ax.set_xticklabels(mat.columns, rotation=35, ha="right")

    ax.set_yticks(np.arange(mat.shape[0]))
    if ax is axes[0]:
        ax.set_yticklabels([f"p={int(p)}%" for p in probabilities])
        ax.set_ylabel("Probability scenario (%)\n(top→bottom: p=17, 33, 50, 67, 83)")
    else:
        ax.set_yticklabels([])

    for i in range(mat.shape[0]):
        for j in range(mat.shape[1]):
            ax.text(j, i, f"{mat.values[i, j]:.2f}", ha="center", va="center", fontsize=9)

cax = fig.add_axes([0.92, 0.18, 0.015, 0.64])
cbar = fig.colorbar(im, cax=cax)
cbar.set_label("Cumulative effect on RCB by 2100 (% of RCB)\n(+ preserves; − consumes)")

fig.suptitle("G All cases under 2.0 ℃ scenario", y=0.98, fontsize=16)
fig.subplots_adjust(left=0.16, right=0.90, top=0.88, bottom=0.22, wspace=0.25)
savefig(fig, "FigC_heatmap_RCB_min_med_max.png")
plt.show()

print("\nAll outputs saved to:", OUT_DIR)
